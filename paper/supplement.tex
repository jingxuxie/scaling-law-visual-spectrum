\section{Notation and Theorem Scope}

Let $z\sim\mathcal N(0,\Sigma)$, $y=\langle z,w^\star\rangle+\xi$, and $\E\xi^2=\sigma^2$.  We distinguish three levels of claims.

\begin{enumerate}
 \item The second-moment and EMA-tracking results apply to online RMSProp/Adam under the stated slow-drift and effective-window assumptions.
 \item The sharp risk filters are proved for a fixed or frozen positive preconditioner $P$.  They also extend coordinatewise to aligned online dynamics when every $P_t$ commutes with $\Sigma$ and is uniformly comparable to a fixed visible profile.
 \item For general noncommuting online feature systems, the visible-spectrum theorem is an instantaneous spectral statement; the experiments test whether its diagnostic predicts actual training behavior.
\end{enumerate}

Constants hidden by $\as$ are independent of the asymptotic variables $M,N,C$ but may depend on fixed spectral, source, and tracking constants.

\section{Second Moments and Raw EMA Tracking}

Let $e_t=w_t-w^\star$ and
\[
 g_t=(\langle z_t,e_t\rangle-\xi_t)z_t,
 \qquad c_t=\sigma^2+\|e_t\|_\Sigma^2.
\]
Let $\mathcal F_t$ denote the history before drawing $(z_t,\xi_t)$.

\begin{lemma}[Conditional second moments]
\label{lem:supp-second}
For every predictable iterate and coordinate $j$,
\[
 \nu_{t,j}:=\E[g_{t,j}^2\mid\mathcal F_t]
 =c_t\Sigma_{jj}+2(\Sigma e_t)_j^2,
\]
and $c_t\Sigma_{jj}\le\nu_{t,j}\le3c_t\Sigma_{jj}$.
\end{lemma}

\begin{proof}
Set $U_t=\langle z_t,e_t\rangle-\xi_t$ and $Z_{t,j}=z_{t,j}$.  Conditionally on $\mathcal F_t$, $(U_t,Z_{t,j})$ is centered Gaussian with variances $c_t$ and $\Sigma_{jj}$ and covariance $(\Sigma e_t)_j$.  Isserlis' identity gives
\[
 \E[U_t^2Z_{t,j}^2\mid\mathcal F_t]
 =c_t\Sigma_{jj}+2(\Sigma e_t)_j^2.
\]
Cauchy--Schwarz in the $\Sigma$ inner product yields $(\Sigma e_t)_j^2\le\|e_t\|_\Sigma^2\Sigma_{jj}\le c_t\Sigma_{jj}$.
\end{proof}

Raw RMSProp uses
\[
 v_{t,j}=\sum_{s<t}\alpha_{t,s}g_{s,j}^2,
 \qquad \alpha_{t,s}=(1-\beta_2)\beta_2^{t-1-s}.
\]
Define the predictable target and scalar residual EMA
\[
 m_{t,j}=\sum_{s<t}\alpha_{t,s}\nu_{s,j},
 \qquad
 d_t=\sum_{s<t}\alpha_{t,s}c_s.
\]
Lemma~\ref{lem:supp-second} gives
\begin{equation}
\label{eq:population-ema}
 d_t\Sigma_{jj}\le m_{t,j}\le3d_t\Sigma_{jj}.
\end{equation}

The squared gradient is a fourth-order Gaussian-chaos quantity.  A standard hypercontractivity argument gives universal $C,c>0$ such that
\begin{equation}
\label{eq:subweibull}
 \Pr(g_{t,j}^2\ge u\nu_{t,j}\mid\mathcal F_t)
 \le2\exp(-c\sqrt{u}),\qquad u\ge1,
\end{equation}
and $\E[(g_{t,j}^2-\nu_{t,j})^2\mid\mathcal F_t]\le C\nu_{t,j}^2$.

Define the leverage
\[
 \Lambda_{t,j}=\max_{s<t}\frac{\alpha_{t,s}\nu_{s,j}}{m_{t,j}},
 \qquad \Lambda=\max_{t\le N,j\le M}\Lambda_{t,j}.
\]

\begin{theorem}[Raw EMA tracking]
\label{thm:raw-ema}
Fix $\tau\in(0,1/2)$ and failure probability $\delta\in(0,1)$.  There is a universal constant $c_0>0$ such that if
\[
 \Lambda\le \frac{c_0\tau^2}{\log^2(2MN/\delta)},
\]
then with probability at least $1-\delta$,
\[
 |v_{t,j}-m_{t,j}|\le\tau m_{t,j}
\]
simultaneously over $t\le N,j\le M$.  Consequently,
\[
 (1-\tau)d_t\Sigma_{jj}
 \le v_{t,j}
 \le3(1+\tau)d_t\Sigma_{jj}.
\]
\end{theorem}

\begin{proof}
Write
\[
 v_{t,j}-m_{t,j}=\sum_{s<t}\alpha_{t,s}(g_{s,j}^2-\nu_{s,j}),
\]
a martingale-difference sum.  By \eqref{eq:subweibull}, each summand has conditional sub-Weibull-$1/2$ norm at most $C\alpha_{t,s}\nu_{s,j}\le C\Lambda m_{t,j}$.  Its predictable quadratic variation is bounded by
\[
 C\sum_{s<t}\alpha_{t,s}^2\nu_{s,j}^2
 \le C\Lambda m_{t,j}^2.
\]
Truncate summands at $c_1\tau m_{t,j}/\log(2MN/\delta)$.  The sub-Weibull tail and leverage condition make the total truncation-failure probability at most $\delta/(2MN)$.  Conditional Freedman's inequality for the truncated martingale gives deviation at most $\tau m_{t,j}$ with failure probability at most $\delta/(2MN)$.  A union bound over $(t,j)$ proves the first statement; combining with \eqref{eq:population-ema} proves the second.
\end{proof}

If $c_s$ changes by at most a factor $\chi$ across the active EMA window, then $\Lambda\lesssim\chi(1-\beta_2)$ up to the exponentially small old-history tail.  Thus an effective window $(1-\beta_2)^{-1}\gtrsim\chi\tau^{-2}\log^2(2MN/\delta)$ suffices.

\section{Fixed Preconditioning as Transformed SGD}

Let $P\succ0$ be fixed.  Write $w=P^{1/2}u$ and $\widetilde z=P^{1/2}z$.  Preconditioned SGD
\[
 w_{t+1}=w_t-\gamma_tP(\langle z_t,w_t\rangle-y_t)z_t
\]
is exactly ordinary SGD in $u$:
\[
 u_{t+1}=u_t-\gamma_t(\langle\widetilde z_t,u_t\rangle-y_t)\widetilde z_t.
\]
The transformed covariance and target are
\[
 \widetilde\Sigma=P^{1/2}\Sigma P^{1/2},
 \qquad
 u^\star=P^{-1/2}w^\star.
\]
This exact change of variables is the basis of all sharp risk statements.

\section{Visible Spectral Comparability}

Assume
\begin{equation}
\label{eq:supp-visible}
 c(\Sigma^\theta+\rho I)^{-1/2}
 \pre P\pre
 C(\Sigma^\theta+\rho I)^{-1/2}.
\end{equation}

\begin{theorem}[Effective spectrum]
The ordered eigenvalues $\widetilde\mu_i$ of $\widetilde\Sigma$ obey
\[
 c\lambda_i(\lambda_i^\theta+\rho)^{-1/2}
 \le\widetilde\mu_i\le
 C\lambda_i(\lambda_i^\theta+\rho)^{-1/2}.
\]
\end{theorem}

\begin{proof}
$P^{1/2}\Sigma P^{1/2}$ and $\Sigma^{1/2}P\Sigma^{1/2}$ have the same nonzero eigenvalues.  Congruence of \eqref{eq:supp-visible} by $\Sigma^{1/2}$ and the min--max principle give the result because $(\Sigma^\theta+\rho I)^{-1/2}$ commutes with $\Sigma$.
\end{proof}

Suppose $\lambda_i\asymp i^{-a}$.  The damping knee satisfies $\lambda_i^\theta\asymp\rho$, so $i_\rho\asymp\rho^{-1/(a\theta)}$.  For $i\lesssim i_\rho$, $\widetilde\mu_i\asymp i^{-a(1-\theta/2)}$; for $i\gtrsim i_\rho$, $\widetilde\mu_i\asymp\rho^{-1/2}i^{-a}$.  The sample-size knee is $n_\rho\asymp\rho^{-(1/\theta-1/2)}$.  Solving $n\widetilde\mu_i\asymp1$ gives the learned-mode formula in the main paper.

\section{Source Stability}

If $P=f(\Sigma)$, the eigenvectors are unchanged and
\[
 \widetilde\mu_i=\lambda_if(\lambda_i),
 \qquad
 \langle v_i,u^\star\rangle=f(\lambda_i)^{-1/2}\langle v_i,w^\star\rangle.
\]
Therefore transformed source energies are exactly invariant:
\[
 \widetilde s_i
 =\widetilde\mu_i\langle v_i,u^\star\rangle^2
 =\lambda_i\langle v_i,w^\star\rangle^2=s_i.
\]

If $\Sigma$ and $P$ share an invariant block decomposition, source mass is exactly preserved in every block.  If eigenvalues and filters are comparable within a block, the band-averaged source exponent is unchanged.  Globally mixed cases require an empirical transformed-source diagnostic.

\section{Sharp Risk Filters}

For geometric step decay, transformed SGD has approximation, bias, and variance filters
\begin{align}
 A_M&=\sum_{i>M}\widetilde s_i,\\
 B(n,M)&=\sum_{i\le M}\frac{\widetilde s_i}{1+n\widetilde\mu_i},\\
 V(n,M)&=\frac1{\Neff}\sum_{i\le M}\min\{1,n^2\widetilde\mu_i^2\}.
\end{align}
These follow from the one-pass SGD bounds of \citeauthor{lin2024scaling} (\citeyear{lin2024scaling}) after the transformed-feature reduction.

Let $K=\#\{i:\widetilde\mu_i\gtrsim n^{-1}\}$.  Assume on the active segment $\widetilde\mu_i\asymp i^{-\alpha}$ and $\widetilde s_i\asymp i^{-b}$.

\begin{theorem}[Matching spectral sums]
If $1<b<\alpha+1$ and $K<M$, then $B(n,M)\asymp K^{1-b}$.  If $\alpha>1/2$, then $V(n,M)\asymp\min\{M,K\}/\Neff$.  Also $A_M\asymp M^{1-b}$.
\end{theorem}

\begin{proof}
Using $(1+x)^{-1}\asymp\min\{1,x^{-1}\}$,
\[
 B(n,M)\asymp
 \frac1n\sum_{i\le K}\frac{\widetilde s_i}{\widetilde\mu_i}
 +\sum_{K<i\le M}\widetilde s_i.
\]
The tail is $\asymp K^{1-b}$, while the head is
\[
 \frac1n\sum_{i\le K}i^{\alpha-b}
 \asymp\frac{K^{\alpha-b+1}}n\asymp K^{1-b}.
\]
For variance, the first $\min\{M,K\}$ terms give the lower bound.  When $K<M$,
\[
 n^2\sum_{i>K}\widetilde\mu_i^2\asymp n^2K^{1-2\alpha}\asymp K
\]
provided $\alpha>1/2$.  Two-slope damped spectra follow by splitting at the knee.
\end{proof}

\section{Compute-Optimal Allocation}

For fixed $N$, minimize $K^{1-b}+K/N$ subject to $K\le N^{1/\alpha}$.  The unconstrained optimum is $K\asymp N^{1/b}$.  If $b\le\alpha$, the time cap is active and the $N$-dependent risk is $N^{-(b-1)/\alpha}$.  If $b>\alpha$, $K\asymp N^{1/b}$ and bias and variance are both $N^{-(b-1)/b}$.  Balancing with $M^{1-b}$ under $C=MN$ gives
\[
 M_\star\asymp C^{1/(m+1)},\quad
 N_\star\asymp C^{m/(m+1)},\quad
 K_\star\asymp C^{1/(m+1)},
\]
where $m=\max\{\alpha,b\}$, and risk exponent $(b-1)/(m+1)$.

\section{Adam Momentum and AdamW}

For a fixed coordinate with effective eigenvalue $\mu$, let $x=\gamma\mu$.  Population Adam momentum gives
\[
 e_{t+1}=[1+\beta_1-(1-\beta_1)x]e_t-\beta_1e_{t-1}.
\]
The characteristic root near one is $1-x+O_{\beta_1}(x^2)$; therefore the learning time remains $x^{-1}$ and momentum does not change the spectral power.

With AdamW decay $\delta$, coordinate error satisfies
\[
 e_{t+1}=[1-\gamma(\mu+\delta)]e_t-\gamma\delta w^\star.
\]
With $w_0=0$ and $r=1-\gamma(\mu+\delta)$,
\[
 e_N=-\left[\frac\delta{\mu+\delta}+\frac\mu{\mu+\delta}r^N\right]w^\star.
\]
Thus $\delta/(\mu+\delta)$ is an irreducible shrinkage floor and active modes require $\mu\gtrsim\max\{n^{-1},\delta\}$.  If $\delta(C)=C^{-s}$, the cap is $K\le C^{s/\alpha}$.  Comparing it with the no-decay optimum proves the AdamW exponent in the main paper.

\section{Feature-Map Results}

\paragraph{Band-limited mixing.}
Partition indices into bands $B_\ell$ with $\kappa^{-1}\Lambda_\ell\le\lambda_i\le\kappa\Lambda_\ell$.  If rotations are block diagonal across these bands, every coordinate variance in a band is comparable to $\Lambda_\ell$.  Hence the diagonal preconditioner is within constants of $(\Lambda_\ell+\rho)^{-1/2}I$ on the block, yielding $\widetilde\mu_i\asymp_\kappa\lambda_i(\lambda_i+\rho)^{-1/2}$.

\paragraph{Flat coordinates.}
If $d_{\min}\le\Sigma_{jj}\le d_{\max}$ and $d_{\max}/d_{\min}\le\kappa$, then $P$ is scalar-comparable, and every effective eigenvalue is within constants of the original eigenvalue.  The same exponent conclusion holds when the diagonal-preconditioner condition number is subpolynomial in dimension.

\paragraph{Gaussian sketches.}
For $s_j\sim\mathcal N(0,I/D)$, $d_j=s_j^\top Hs_j$ and $\E d_j=\tr(H)/D$.  Weighted chi-square concentration gives
\[
 \Pr\left(|d_j-\tr(H)/D|\ge t\tr(H)/D\right)
 \le2\exp\{-c\min(t^2r_{\rm eff},tr_{\rm st})\},
\]
where $r_{\rm eff}=\tr(H)^2/\tr(H^2)$ and $r_{\rm st}=\tr(H)/\|H\|_{\rm op}$.  When these ranks are sufficiently large, a union bound gives a flat diagonal.  Otherwise, measured $\qeff$ determines the finite-dimensional visible regime.

\section{Diagnostic Consistency}

Suppose on a rank window $i\in[L,U]$,
\[
 \lambda_i=c_\lambda i^{-a}e^{\varepsilon_i},
 \qquad
 \widetilde\mu_i=c_\mu i^{-\alpha}e^{\delta_i},
 \qquad |\varepsilon_i|,|\delta_i|\le\eta.
\]
Least-squares slope errors are $O(\eta/\log(U/L))$.  If $a$ is bounded away from zero, the same order controls $|\widehat q-q|$, where $q=1-\alpha/a$.

\section{Experimental Details}

All stochastic runs use mini-batch Gaussian regression, geometric checkpoints, and optimizer-specific learning-rate grids.  Targets are constructed in the population eigenbasis with source energies $i^{-b}$.  We average over independent trials and report the learning rate with the best final mean risk.  The code logs the effective covariance induced by the second-moment preconditioner and fits $\qeff$ on a central rank window.

The visible-profile run uses dimension $2048$, $(a,b)=(1.5,1.4)$, five trials, and $\theta\in\{0,.25,.5,.75,1\}$.  The wide-LR random-feature run uses feature dimension $256$, ambient dimension $2048$, 16,000 updates, ten trials, and adaptive learning rates from $3\times10^{-4}$ to $10^{-1}$.  Compute-optimal experiments optimize the sharp proxy over $(M,N,K)$; AdamW experiments add the compute-dependent cap.

Aligned and flat coordinates preserve the source exponent to numerical precision.  Band-limited rotations preserve source mass exactly by block and show small finite-band slope deviations.  Haar rotations can change coordinatewise source slopes, motivating explicit transformed-source reporting in globally mixed experiments.
